cmake_minimum_required(VERSION 3.16)
project(netguard VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.7 REQUIRED)
add_library(netguard_core src/netguard.cpp)
target_include_directories(netguard_core PUBLIC include)
target_link_libraries(netguard_core PUBLIC Threads::Threads nlohmann_json::nlohmann_json)
target_compile_options(netguard_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(netguard src/main.cpp)
target_link_libraries(netguard PRIVATE netguard_core)
enable_testing()
add_executable(engine_test tests/engine_test.cpp)
target_link_libraries(engine_test PRIVATE netguard_core)
add_test(NAME engine_test COMMAND engine_test)
